# =================================================================
# 错误追踪配置（Sentry）
# =================================================================
# 启用 Sentry 错误追踪：
# 1. 在 https://sentry.io 创建账号
# 2. 为 Next.js 创建新项目
# 3. 按下方步骤完成配置

# 步骤 1：Sentry DSN（必填）
# 路径：Settings > Projects > [Your Project] > Client Keys (DSN)

NEXT_PUBLIC_SENTRY_DSN=    # 示例: https://****@****.ingest.sentry.io/****


# 步骤 2：组织与项目信息
# 路径：Settings > Organization > General Settings

NEXT_PUBLIC_SENTRY_ORG=  # 示例: acme-corp
NEXT_PUBLIC_SENTRY_PROJECT=  # 示例: nextjs-dashboard


# 步骤 3：Sentry Auth Token

# Sentry 可使用 source maps 为错误提供可读堆栈，因此需要 auth token。
# 更多信息: https://docs.sentry.io/platforms/javascript/guides/nextjs/manual-setup/#step-4-add-readable-stack-traces-with-source-maps-optional

SENTRY_AUTH_TOKEN=    # 示例: sntrys_************************************


# 步骤 4：环境控制（可选）
# 设置为 'true' 可在开发环境禁用 Sentry

NEXT_PUBLIC_SENTRY_DISABLED=false
